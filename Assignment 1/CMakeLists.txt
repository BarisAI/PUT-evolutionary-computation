cmake_minimum_required(VERSION 3.16)
project(assignment1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(assignment1 src/main.cpp)
add_executable(checker src/checker.cpp)
add_executable(heuristic_tests src/tests.cpp)
enable_testing()
add_test(NAME heuristic_tests COMMAND heuristic_tests)
foreach(target assignment1 checker heuristic_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
