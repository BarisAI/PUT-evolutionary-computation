cmake_minimum_required(VERSION 3.16)
project(assignment3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(assignment3 src/main.cpp)
add_executable(local_search_tests src/tests.cpp)
add_executable(checker src/checker.cpp)
enable_testing()
add_test(NAME local_search_tests COMMAND local_search_tests)
foreach(target assignment3 local_search_tests checker)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /O2)
  else()
    target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -Wpedantic)
  endif()
endforeach()
