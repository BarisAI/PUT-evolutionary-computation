cmake_minimum_required(VERSION 3.16)
project(assignment5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(assignment5 src/main.cpp)
add_executable(lm_tests src/tests.cpp)
add_executable(checker src/checker.cpp)
enable_testing()
add_test(NAME lm_tests COMMAND lm_tests)
foreach(target assignment5 lm_tests checker)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /O2)
 else()
  target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Wpedantic)
 endif()
endforeach()
