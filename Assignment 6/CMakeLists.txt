cmake_minimum_required(VERSION 3.16)
project(assignment6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(assignment6 src/main.cpp)
add_executable(ils_tests src/tests.cpp)
add_executable(checker src/checker.cpp)
enable_testing()
add_test(NAME ils_tests COMMAND ils_tests)
foreach(target assignment6 ils_tests checker)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /O2)
 else()
  target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Wpedantic)
 endif()
endforeach()
