cmake_minimum_required(VERSION 3.16)
project(assignment8 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(assignment8 src/main.cpp)
add_executable(assignment8_tests src/tests.cpp)
add_executable(assignment8_checker src/checker.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(t assignment8 assignment8_tests assignment8_checker)
    target_compile_options(${t} PRIVATE -O2 -Wall -Wextra -Wpedantic)
  endforeach()
endif()
enable_testing()
add_test(NAME greedy_local_optimum COMMAND assignment8_tests)
