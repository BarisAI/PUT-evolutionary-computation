cmake_minimum_required(VERSION 3.16)
project(assignment7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(assignment7 src/main.cpp)
add_executable(assignment7_tests src/tests.cpp)
add_executable(assignment7_checker src/checker.cpp)
foreach(target assignment7 assignment7_tests assignment7_checker)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /O2)
    else()
        target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Wpedantic)
    endif()
endforeach()
enable_testing()
add_test(NAME assignment7_property_tests COMMAND assignment7_tests)
